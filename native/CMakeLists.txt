cmake_minimum_required(VERSION 3.20)
project(SteamSDADevice LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MH "${CMAKE_CURRENT_SOURCE_DIR}/vendor/minhook")
add_library(minhook STATIC "${MH}/src/buffer.c" "${MH}/src/hook.c"
  "${MH}/src/trampoline.c" "${MH}/src/hde/hde64.c")
target_include_directories(minhook PRIVATE "${MH}/include")
add_library(SteamSDADevice SHARED device.cpp)
target_include_directories(SteamSDADevice PRIVATE "${MH}/include")
target_compile_definitions(SteamSDADevice PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(SteamSDADevice PRIVATE /W4 /utf-8 /EHsc)
target_link_options(SteamSDADevice PRIVATE /OPT:NOICF)
target_link_libraries(SteamSDADevice PRIVATE minhook ole32)
add_executable(SteamSDA.DeviceFixture fixture.cpp)
